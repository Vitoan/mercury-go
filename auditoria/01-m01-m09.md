```javascript
// pedidos_store.js:12-25
export function con_cambios_sin_enviar(pedido) {
  if (!pedido) return pedido;
  const offline = use_offline_store();
  const cambios = (offline.cola || []).filter((c) => c.pedido_id == pedido.id && c.estado !== 'enviada');
  if (cambios.length === 0) return pedido;
  const ultimo = cambios[cambios.length - 1];
  const estadoNegocio = ultimo.payload?.estado || MAPA_ESTADO_OFFLINE[ultimo.tipo] || pedido.estado;
  return {
    ...pedido,
    estado: estadoNegocio,
    sin_enviar: true
  };
}
```

  * Al desactivar el modo avión, la cola sincroniza automáticamente sin duplicar registros ni dejar tarjetas en estado `'enviada'` falso.

### Puntos donde puede trabarse la demo:
1. **Paso de Biometría:** Si el teléfono de prueba no tiene huella dactilar registrada en los ajustes de Android, el botón de huella no se habilitará; se debe usar el login con credenciales (`Mercury123!` o `Mercury2026!Admin`).
2. **Reasignación de Choferes:** Si un pedido está asignado a un chofer y se ingresa con otro usuario `CHOFER`, la orden no aparecerá en la lista de ruta (debido al aislamiento IDOR corregido en `DEF-12`). Asegurarse de usar el usuario chofer asignado al pedido.
3. **Escaneo QR en Modo Avión:** Google ML Kit requiere haber descargado el módulo de escaneo al menos una vez con conexión antes de operar completamente offline.
4. **Push con Modo Ahorro de Batería:** Algunos fabricantes de Android (Xiaomi, Samsung) pueden demorar las notificaciones push con la pantalla apagada si la app tiene optimización de batería agresiva activada.
